/**
 * Función inteligente: Lee el PropertiesService.
 * Si está vacío (por migración), usa el valor de fábrica y lo inyecta en la bóveda.
 */
function getConfig(key) {
  const propiedades = PropertiesService.getDocumentProperties();
  let valor = propiedades.getProperty(key);

  if (valor === null || valor === "") {
    valor = DEFAULTS[key]; // Paracaídas activado
    
    if (valor !== undefined) {
      propiedades.setProperty(key, valor.toString()); // Auto-reparación
    }
  }

  // Parseo de tipos para que el código entienda booleanos y números
  if (valor === "true") return true;
  if (valor === "false") return false;
  if (!isNaN(valor) && valor !== "" && key.includes("DIAS")) return parseInt(valor, 10);

  return valor;
}
