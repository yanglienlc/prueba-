function onOpen() {
  SpreadsheetApp.getUi().createMenu('🚀 Expedientes LMD')
    .addItem('⚙️ Panel de Control', 'abrirPanelControl')
    .addItem('📊 Termómetro Consular', 'abrirTermometro')
    .addItem('🔍 Estado consular La Habana', 'ejecutarEstadoConsularManual')
    .addSeparator()
    .addItem('1. Verificar Vencimientos (Manual)', 'ejecutarAuditorManual')
    .addItem('2. Enviar Correos Pendientes', 'ejecutarEnviosManual')
    .addToUi();
}

function abrirPanelControl() {
  const html = HtmlService.createTemplateFromFile('Modal_PanelControl').evaluate()
    .setWidth(550)
    .setHeight(650);
  SpreadsheetApp.getUi().showModalDialog(html, '⚙️ Panel de Configuración');
}

function abrirTermometro() {
  const html = HtmlService.createTemplateFromFile('Modal_Termometro').evaluate()
    .setWidth(450)
    .setHeight(500);
  SpreadsheetApp.getUi().showModalDialog(html, '📊 Inteligencia de Negocio');
}

// Puentes para capturar el resultado y mostrar la ventana emergente
function ejecutarAuditorManual() {
  const resultado = trabajadorNocturno(); 
  mostrarReporte("Verificación de Vencimientos", resultado);
}

function ejecutarEnviosManual() {
  const resultado = procesarNotificacionesLMD(); 
  mostrarReporte("Envío de Notificaciones", resultado);
}

function mostrarReporte(titulo, resultado) {
  if (!resultado || resultado.cancelado) return; 
  
  const plantilla = HtmlService.createTemplateFromFile('Modal_Reporte');
  plantilla.titulo = titulo;
  plantilla.datos = JSON.stringify(resultado); 
  const html = plantilla.evaluate().setWidth(500).setHeight(550);
  SpreadsheetApp.getUi().showModalDialog(html, '📊 Reporte del Sistema');
}
